cmake_minimum_required(VERSION 3.25)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(RuToolPortableCrypto LANGUAGES C CXX)
    option(BUILD_TESTING "Build private developer checks" OFF)
    include(CTest)
endif()
set(rutool_mbedtls "${CMAKE_CURRENT_SOURCE_DIR}/../third_party/mbedtls-3.6.7")
set(rutool_mbed_sources aes asn1parse asn1write base64 bignum bignum_core
    bignum_mod bignum_mod_raw block_cipher cipher cipher_wrap constant_time
    ecdsa ecp ecp_curves gcm md oid pkcs5 platform platform_util sha256)
set(rutool_mbed_files)
foreach(name IN LISTS rutool_mbed_sources)
    list(APPEND rutool_mbed_files "${rutool_mbedtls}/library/${name}.c")
endforeach()
add_library(rutool_mbedcrypto STATIC ${rutool_mbed_files})
set_target_properties(rutool_mbedcrypto PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(rutool_mbedcrypto SYSTEM PUBLIC "${rutool_mbedtls}/include")
target_include_directories(rutool_mbedcrypto PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(rutool_mbedcrypto PUBLIC MBEDTLS_CONFIG_FILE="rutool_mbedtls_config.h")
add_library(rutool_portable_crypto STATIC PortableCrypto.cpp)
set_target_properties(rutool_portable_crypto PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_features(rutool_portable_crypto PUBLIC cxx_std_20)
target_include_directories(rutool_portable_crypto PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(rutool_portable_crypto PUBLIC rutool_mbedcrypto)
if(BUILD_TESTING)
    add_executable(rutool_portable_crypto_tests ../tests/support/PortableCryptoTests.cpp)
    target_link_libraries(rutool_portable_crypto_tests PRIVATE rutool_portable_crypto)
    target_include_directories(rutool_portable_crypto_tests SYSTEM PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../third_party")
    target_compile_definitions(rutool_portable_crypto_tests PRIVATE RUTOOL_CRYPTO_FIXTURES="${CMAKE_CURRENT_SOURCE_DIR}/../tests/fixtures")
    add_test(NAME portable_crypto_vectors COMMAND rutool_portable_crypto_tests)
endif()
